    <script>
        // Casino Simulator Logic
        function runCasinoSim() {
            let casinoWins = 0;
            let playerWins = 0;
            const houseEdgeChance = 0.527; // 52.7% chance for the casino in American roulette
            const totalSpins = 1000;

            for (let i = 0; i < totalSpins; i++) {
                if (Math.random() < houseEdgeChance) {
                    casinoWins++;
                } else {
                    playerWins++;
                }
            }

            document.getElementById('casinoWins').innerText = casinoWins;
            document.getElementById('casinoLosses').innerText = playerWins;
            
            const netProfit = (casinoWins - playerWins) * 10;
            const profitEl = document.getElementById('casinoProfit');
            profitEl.innerText = `$${netProfit.toLocaleString()}`;
            
            if (netProfit > 0) {
                profitEl.className = "text-2xl font-extrabold text-emerald-400";
            } else {
                profitEl.className = "text-2xl font-extrabold text-rose-400";
            }
        }

        // Trading Simulator Logic
        let equityChart = null;

        function updateSliders() {
            const winRate = document.getElementById('winRateInput').value;
            const risk = document.getElementById('riskInput').value;
            const reward = document.getElementById('rewardInput').value;

            document.getElementById('winRateVal').innerText = `${winRate}%`;
            document.getElementById('riskVal').innerText = `$${risk}`;
            document.getElementById('rewardVal').innerText = `${reward}x (1:${reward})`;
        }

        function runTradingSimulation(tradesCount) {
            const winRate = parseInt(document.getElementById('winRateInput').value) / 100;
            const risk = parseInt(document.getElementById('riskInput').value);
            const rewardMult = parseFloat(document.getElementById('rewardInput').value);
            const reward = risk * rewardMult;

            let balance = 10000; // Starting virtual balance
            const balanceHistory = [balance];
            const labels = ['Start'];

            let wins = 0;
            let maxStreak = 0;
            let currentStreak = 0;

            for (let i = 1; i <= tradesCount; i++) {
                const win = Math.random() < winRate;
                if (win) {
                    balance += reward;
                    wins++;
                    currentStreak = 0;
                } else {
                    balance -= risk;
                    currentStreak++;
                    if (currentStreak > maxStreak) {
                        maxStreak = currentStreak;
                    }
                }
                balanceHistory.push(balance);
                labels.push(`T${i}`);
            }

            // Update Statistics
            document.getElementById('statWinRate').innerText = `${Math.round((wins / tradesCount) * 100)}%`;
            document.getElementById('statStreak').innerText = `${maxStreak} Losses`;
            const netGrowth = balance - 10000;
            const growthEl = document.getElementById('statGrowth');
            growthEl.innerText = (netGrowth >= 0 ? '+' : '') + `$${netGrowth}`;
            growthEl.className = `text-sm font-bold ${netGrowth >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;

            // Chart Rendering
            if (equityChart) {
                equityChart.destroy();
            }

            const ctx = document.getElementById('equityChart').getContext('2d');
            equityChart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: 'Portfolio Equity ($)',
                        data: balanceHistory,

                    }]
